```kotlin
            JSONArray(repeatDays).let { arr ->

            }
        } catch (e: Exception) {
            emptyList()
        }
    }
```

```kotlin
object NotificationHelper {
    private const val CHANNEL_REMINDER = "reminder_channel"
    private const val CHANNEL_TOMATO = "tomato_channel"
    private const val CHANNEL_ALARM = "alarm_channel"

    fun createNotificationChannels(context: Context) {
        val channels = listOf(
            NotificationChannel(
                CHANNEL_REMINDER, "日常提醒",
                NotificationManager.IMPORTANCE_HIGH
            ).apply { description = "喝水、吃药等日常提醒" },
            NotificationChannel(
                CHANNEL_TOMATO, "番茄钟",
                NotificationManager.IMPORTANCE_DEFAULT
            ).apply { description = "番茄钟专注和休息提醒" },
            NotificationChannel(
                CHANNEL_ALARM, "闹钟",
                NotificationManager.IMPORTANCE_HIGH
            ).apply { description = "闹钟响铃提醒" }
        )
        val manager = context.getSystemService(NotificationManager::class.java)
        channels.forEach { manager.createNotificationChannel(it) }
    }

    fun showReminderNotification(context: Context, title: String, message: String) {
        val intent = Intent(context, MainActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK or Intent.FLAG_ACTIVITY_CLEAR_TASK
        }
        val pendingIntent = PendingIntent.getActivity(
            context, 0, intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        val notification = NotificationCompat.Builder(context, CHANNEL_REMINDER)
            .setSmallIcon(R.drawable.ic_notification)
            .setContentTitle(title)
            .setContentText(message)
            .setPriority(NotificationCompat.PRIORITY_HIGH)
            .setContentIntent(pendingIntent)
            .setAutoCancel(true)
            .build()
        NotificationManagerCompat.from(context).notify(
            System.currentTimeMillis().toInt(),
            notification
        )
    }
}
```

### 3.2 AlarmReceiver（闹钟广播接收器）

```kotlin
class AlarmReceiver : BroadcastReceiver() {

    override fun onReceive(context: Context, intent: Intent) {
        val action = intent.action ?: return
        when {
            action.startsWith(ACTION_REMINDER) -> handleReminder(context, intent)
            action.startsWith(ACTION_ALARM) -> handleAlarm(context, intent)
            action == Intent.ACTION_BOOT_COMPLETED -> handleBootCompleted(context)
        }
    }

    private fun handleReminder(context: Context, intent: Intent) {
        val title = intent.getStringExtra(EXTRA_TITLE) ?: "提醒"
        val message = intent.getStringExtra(EXTRA_MESSAGE) ?: "时间到了！"
        NotificationHelper.showReminderNotification(context, title, message)
    }

    private fun handleAlarm(context: Context, intent: Intent) {
        val alarmId = intent.getLongExtra(EXTRA_ALARM_ID, -1L)
        val label = intent.getStringExtra(EXTRA_LABEL) ?: "闹钟"
        // 启动闹钟响铃 Activity
        val alarmIntent = Intent(context, AlarmActivity::class.java).apply {
            flags = Intent.FLAG_ACTIVITY_NEW_TASK
            putExtra(EXTRA_ALARM_ID, alarmId)
            putExtra(EXTRA_LABEL, label)
        }
        context.startActivity(alarmIntent)
    }

    private fun handleBootCompleted(context: Context) {
        // 开机后重新注册所有启用的提醒和闹钟
        CoroutineScope(Dispatchers.IO).launch {
            val db = AppDatabase.getInstance(context)
            val enabledReminders = db.reminderDao().getEnabledReminders()
            val enabledAlarms = db.alarmDao().getEnabledAlarms()
            enabledReminders.forEach { AlarmScheduler.scheduleReminder(context, it) }
            enabledAlarms.forEach { AlarmScheduler.scheduleAlarm(context, it) }
        }
    }

    companion object {
        const val ACTION_REMINDER = "com.timer.reminder.ACTION_REMINDER"
        const val ACTION_ALARM = "com.timer.reminder.ACTION_ALARM"
        const val EXTRA_TITLE = "extra_title"
        const val EXTRA_MESSAGE = "extra_message"
        const val EXTRA_ALARM_ID = "extra_alarm_id"
        const val EXTRA_LABEL = "extra_label"
    }
}
```

### 3.3 AlarmScheduler（闹钟调度器）

```kotlin
object AlarmScheduler {

    /** 注册日常提醒闹钟 */
    fun scheduleReminder(context: Context, reminder: ReminderEntity) {
        val alarmManager = context.getSystemService(AlarmManager::class.java)
        val intent = Intent(context, AlarmReceiver::class.java).apply {
            action = AlarmReceiver.ACTION_REMINDER + reminder.id
            putExtra(AlarmReceiver.EXTRA_TITLE, reminder.title)
            putExtra(AlarmReceiver.EXTRA_MESSAGE, "${reminder.type}时间到了！")
        }
        val pendingIntent = PendingIntent.getBroadcast(
            context, reminder.id.toInt(), intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val calendar = Calendar.getInstance().apply {
            set(Calendar.HOUR_OF_DAY, reminder.hour)
            set(Calendar.MINUTE, reminder.minute)
            set(Calendar.SECOND, 0)
            set(Calendar.MILLISECOND, 0)
            if (before(Calendar.getInstance())) {
                add(Calendar.DAY_OF_YEAR, 1)
            }
        }

        when (reminder.repeatType) {
            "每天" -> alarmManager.setRepeating(
                AlarmManager.RTC_WAKEUP,
                calendar.timeInMillis,
                AlarmManager.INTERVAL_DAY,
                pendingIntent
            )
            "工作日" -> { /* 参见 3.4 工作日调度逻辑 */ }
            "自定义" -> { /* 参见 3.4 自定义调度逻辑 */ }
        }
    }

    /** 注册一次性闹钟 */
    fun scheduleAlarm(context: Context, alarm: AlarmEntity) {
        val alarmManager = context.getSystemService(AlarmManager::class.java)
        val intent = Intent(context, AlarmReceiver::class.java).apply {
            action = AlarmReceiver.ACTION_ALARM + alarm.id
            putExtra(AlarmReceiver.EXTRA_ALARM_ID, alarm.id)
            putExtra(AlarmReceiver.EXTRA_LABEL, alarm.label)
        }
        val pendingIntent = PendingIntent.getBroadcast(
            context, (10000 + alarm.id).toInt(), intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )

        val calendar = Calendar.getInstance().apply {
            set(Calendar.HOUR_OF_DAY, alarm.hour)
            set(Calendar.MINUTE, alarm.minute)
            set(Calendar.SECOND, 0)
            if (before(Calendar.getInstance())) {
                add(Calendar.DAY_OF_YEAR, 1)
            }
        }

        // 每天重复
        alarmManager.setRepeating(
            AlarmManager.RTC_WAKEUP,
            calendar.timeInMillis,
            AlarmManager.INTERVAL_DAY,
            pendingIntent
        )
    }

    /** 取消闹钟 */
    fun cancelReminder(context: Context, reminderId: Long) {
        val alarmManager = context.getSystemService(AlarmManager::class.java)
        val intent = Intent(context, AlarmReceiver::class.java).apply {
            action = AlarmReceiver.ACTION_REMINDER + reminderId
        }
        val pendingIntent = PendingIntent.getBroadcast(
            context, reminderId.toInt(), intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        alarmManager.cancel(pendingIntent)
    }

    fun cancelAlarm(context: Context, alarmId: Long) {
        val alarmManager = context.getSystemService(AlarmManager::class.java)
        val intent = Intent(context, AlarmReceiver::class.java).apply {
            action = AlarmReceiver.ACTION_ALARM + alarmId
        }
        val pendingIntent = PendingIntent.getBroadcast(
            context, (10000 + alarmId).toInt(), intent,
            PendingIntent.FLAG_UPDATE_CURRENT or PendingIntent.FLAG_IMMUTABLE
        )
        alarmManager.cancel(pendingIntent)
    }
}
```

```kotlin
class TomatoForegroundService : Service() {

    private lateinit var timer: CountDownTimer
    private var remainingSeconds = 0L

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        val action = intent?.action ?: return START_NOT_STICKY

        when (action) {
            ACTION_START -> {
                val duration = intent.getIntExtra(EXTRA_DURATION, 25) * 60L
                startForeground(NOTIFICATION_ID, createNotification())
                startTimer(duration)
            }
            ACTION_PAUSE -> timer.cancel()
            ACTION_RESUME -> startTimer(remainingSeconds)
            ACTION_STOP -> stopSelf()
        }
        return START_STICKY
    }

    private fun startTimer(totalSeconds: Long) {
        remainingSeconds = totalSeconds
        timer = object : CountDownTimer(totalSeconds * 1000, 1000) {
            override fun onTick(millisUntilFinished: Long) {
                remainingSeconds = millisUntilFinished / 1000
                // 发送倒计时更新到 UI（通过 LiveData/Flow）
                TomatoStateHolder.updateTime(remainingSeconds)
            }
            override fun onFinish() {
                // 倒计时完成
                TomatoStateHolder.onTimerComplete()
                stopSelf()
            }
        }.start()
    }

    companion object {
        const val ACTION_START = "com.timer.reminder.TOMATO_START"
        const val ACTION_PAUSE = "com.timer.reminder.TOMATO_PAUSE"
        const val ACTION_RESUME = "com.timer.reminder.TOMATO_RESUME"
        const val ACTION_STOP = "com.timer.reminder.TOMATO_STOP"
        const val EXTRA_DURATION = "extra_duration"
        private const val NOTIFICATION_ID = 1001
    }
}
```

```kotlin
@HiltViewModel
class TomatoViewModel @Inject constructor(
    private val tomatoRepository: TomatoRepository,
    private val appContext: Application
) : ViewModel() {

    data class TomatoState(
        val phase: String = "focus",          // focus / break
        val isRunning: Boolean = false,
        val remainingSeconds: Long = 25 * 60, // 剩余秒数
        val progress: Float = 1f,             // 0~1
        val completedCount: Int = 0,
        val formattedTime: String = "25:00"
    )

    private val _state = MutableStateFlow(TomatoState())
    val state: StateFlow<TomatoState> = _state.asStateFlow()

    private var timerJob: Job? = null

    init {
        loadTodayCompleted()
    }

    fun toggleTimer() {
        if (_state.value.isRunning) {
            pauseTimer()
        } else {
            if (_state.value.remainingSeconds > 0) {
                resumeTimer()
            } else {
                startTimer()
            }
        }
    }

    private fun startTimer() {
        val duration = if (_state.value.phase == "focus") {
            _state.value.remainingSeconds
        } else {
            5 * 60L  // 休息默认5分钟
        }
        _state.update { it.copy(isRunning = true) }
        startForegroundService(duration)
        timerJob = viewModelScope.launch {
            var remaining = duration
            while (remaining > 0) {
                _state.update {
                    it.copy(
                        remainingSeconds = remaining,
                        formattedTime = formatTime(remaining),
                        progress = remaining.toFloat() / duration
                    )
                }
                delay(1000)
                remaining--
            }
            onTimerComplete()
        }
    }

    private fun onTimerComplete() {
        viewModelScope.launch {
            // 保存记录
            tomatoRepository.insertRecord(TomatoRecordEntity(
                focusDuration = 25,
                startTime = System.currentTimeMillis(),
                type = _state.value.phase,
                isCompleted = true,
                date = getTodayDate()
            ))
            // 切换到下一阶段
            val nextPhase = if (_state.value.phase == "focus") "break" else "focus"
            val nextDuration = if (nextPhase == "focus") 25 * 60L else 5 * 60L
            _state.update {
                it.copy(
                    phase = nextPhase,
                    remainingSeconds = nextDuration,
                    isRunning = false,
                    progress = 1f,
                    formattedTime = formatTime(nextDuration)
                )
            }
            loadTodayCompleted()
            NotificationHelper.showTomatoComplete(appContext, nextPhase)
        }
    }

    private fun formatTime(seconds: Long): String {
        val min = seconds / 60
        val sec = seconds % 60
        return "%02d:%02d".format(min, sec)
    }

    private suspend fun loadTodayCompleted() {
        val count = tomatoRepository.getTodayCompletedCount()
        _state.update { it.copy(completedCount = count) }
    }
}
```
